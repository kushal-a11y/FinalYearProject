  <script>
    const userId = "{{ user_id }}";
    const existingPreferences = {{ preferences| tojson | safe }};
    const casteList = {{ castes | tojson | safe }};
    const degreeList = {{ degrees | tojson | safe }};
    const professionList = {{ professions | tojson | safe }};

    const preferenceKeys = ["preferred_age_min", "preferred_age_max", "preferred_gender", "preferred_education", "preferred_profession", "preferred_caste", "preferred_religion", "preferred_residence", "preferred_height_cm"];
    const priorityFeatures = [
      { key: "age", label: "Age" }, { key: "religion", label: "Religion" },
      { key: "caste", label: "Caste" }, { key: "education", label: "Education" }, { key: "profession", label: "Profession" },
      { key: "height", label: "Height" }, { key: "residence", label: "Residence" }
    ];
    function resetForm(){
      document.getElementById("preferencesForm").reset();  // clear form
    }
    function populateDataLists() {
      document.getElementById("casteOptions").innerHTML = casteList.map(x => `<option value="${x}">`).join('');
      document.getElementById("degreeOptions").innerHTML = degreeList.map(x => `<option value="${x}">`).join('');
      document.getElementById("professionOptions").innerHTML = professionList.map(x => `<option value="${x}">`).join('');
    }
    function fillFormFromExisting() {
      if (!existingPreferences || Object.keys(existingPreferences).length === 0) {
        alert("No saved preferences found.");
        return;
      }

      // --- FIX FOR AGE: split "28 - 29" ---
      if (existingPreferences.age) {
        const parts = existingPreferences.age.split("-").map(x => x.trim());
        document.getElementById("preferred_age_min").value = parts[0] ?? "";
        document.getElementById("preferred_age_max").value = parts[1] ?? "";
      }

      // --- Fill the rest normally ---
      preferenceKeys.forEach(id => {
        if (id.startsWith("preferred_age")) return; // skip age, already handled

        const el = document.getElementById(id);
        if (!el) return;

        const key = id.replace("preferred_", "");

        el.value = existingPreferences[id] ?? existingPreferences[key] ?? "";
      });
    }
    function resetToExisting() { fillFormFromExisting(); }
    function useExistingPreferences() { showLoading("Applying saved preferences..."); setTimeout(() => { hideLoading(); showPriorityChoiceOverlay(); }, 700); }
    function showPriorityChoiceOverlay() {
      const box = document.createElement("div"); box.className = "card"; box.style.maxWidth = "420px";
      box.innerHTML = `<h2>Set Priority?</h2><div class="subtitle">Do you want to assign importance values now?</div><div style="display:flex; gap:10px; justify-content:center;"><button class="btn-primary" id="yes">Yes</button><button class="btn-muted" id="no">Skip</button></div>`;
      const overlay = document.createElement("div"); overlay.style.position = "fixed"; overlay.style.inset = "0"; overlay.style.background = "rgba(44,33,38,0.55)"; overlay.style.backdropFilter = "blur(4px)"; overlay.style.display = "flex"; overlay.style.justifyContent = "center"; overlay.style.alignItems = "center"; overlay.style.zIndex = "300"; overlay.style.padding = "20px"; overlay.appendChild(box); document.body.appendChild(overlay);
      box.querySelector("#yes").onclick = () => { overlay.remove(); openPriorityPanel(); };
      box.querySelector("#no").onclick = () => { overlay.remove(); goNonWeighted(); };
    }
    async function submitPreferences() {
      const formData = {};
    
      // ALWAYS include all fields, even if empty
      preferenceKeys.forEach(id => {
        const el = document.getElementById(id);
        formData[id] = (el && el.value !== undefined) ? el.value : "";
      });
    
      showLoading("Saving preferences...");
    
      try {
        await fetch(`/preference/preferences/${userId}`, {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify(formData)
        });
      
        hideLoading();
        showPriorityChoiceOverlay();
      } catch (err) {
        hideLoading();
        alert("Error saving preferences");
      }
    }
    function openPriorityPanel() {
      const g = document.getElementById("priorityGrid"); g.innerHTML = "";
      priorityFeatures.forEach(f => {
        const id = `prio_${f.key}`;
        const item = document.createElement("div"); item.className = "priority-item";
        item.innerHTML = `
          <div class="priority-label">${f.label}</div>
          <div class="slider-wrapper">
            <span id="${id}_val">5</span>
            <input type="range" id="${id}" min="1" max="10" value="5">
          </div>
          <div class="slider-labels"><span>1</span><span>10</span></div>`;
        g.appendChild(item);
        const slider = item.querySelector(`#${id}`);
        const val = item.querySelector(`#${id}_val`);
        function updateValue() { val.textContent = slider.value; val.style.left = `calc(${(slider.value - 1) / 9 * 100}% )`; }
        slider.addEventListener("input", updateValue);
        updateValue();
      });
      document.getElementById("priorityOverlay").style.display = "flex";
    }

    function closePriorityPanel() { document.getElementById("priorityOverlay").style.display = "none"; }
    async function submitPriorities() {
      const p = {}; priorityFeatures.forEach(f => { const el = document.getElementById(`prio_${f.key}`); if (el) p[`${f.key}_priority`] = parseInt(el.value); });
      closePriorityPanel(); showLoading("Saving priorities...");
      try {
        await fetch(`/priority/set-priority/${userId}`, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify(p) });
        showLoading("Calculating weighted matches..."); setTimeout(() => window.location.href = `/user/flatmatch-results/${userId}`, 900);
      } catch { hideLoading(); alert("Error saving priorities"); }
    }

    function goNonWeighted() { showLoading("Fetching matches..."); setTimeout(() => window.location.href = `/user/flatmatch-results/${userId}`, 700); }
    function showLoading(msg) { document.getElementById("loadingTitle").textContent = msg; document.getElementById("loadingOverlay").style.display = "flex"; }
    function hideLoading() { document.getElementById("loadingOverlay").style.display = "none"; }

    document.addEventListener("DOMContentLoaded", () => {
      populateDataLists();
      if (existingPreferences && Object.keys(existingPreferences).length > 0) { document.getElementById("existingCard").style.display = "block"; }
      fillFormFromExisting();
    });
  </script>
